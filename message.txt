<script>
  // Loader
  window.addEventListener('load', () => {
    setTimeout(() => {
      document.getElementById('loader').classList.add('hidden');
    }, 1800);
  });

  // Custom Cursor
  const cursor = document.getElementById('cursor');
  const follower = document.getElementById('cursorFollower');
  let mx = 0, my = 0, fx = 0, fy = 0;

  document.addEventListener('mousemove', e => {
    mx = e.clientX; my = e.clientY;
    cursor.style.left = mx + 'px';
    cursor.style.top = my + 'px';
  });

  function animateFollower() {
    fx += (mx - fx) * 0.12;
    fy += (my - fy) * 0.12;
    follower.style.left = fx + 'px';
    follower.style.top = fy + 'px';
    requestAnimationFrame(animateFollower);
  }
  animateFollower();

  document.querySelectorAll('a, button, .product-btn, .product-card').forEach(el => {
    el.addEventListener('mouseenter', () => {
      cursor.classList.add('hover');
      follower.classList.add('hover');
    });
    el.addEventListener('mouseleave', () => {
      cursor.classList.remove('hover');
      follower.classList.remove('hover');
    });
  });

  // Scroll Fade In
  const fadeEls = document.querySelectorAll('.fade-up');
  const observer = new IntersectionObserver((entries) => {
    entries.forEach(e => {
      if (e.isIntersecting) {
        e.target.classList.add('visible');
      }
    });
  }, { threshold: 0.12 });
  fadeEls.forEach(el => observer.observe(el));

  // Newsletter
  document.querySelector('.newsletter-btn').addEventListener('click', () => {
    const input = document.querySelector('.newsletter-input');
    if (input.value) {
      input.value = '';
      input.placeholder = 'شكراً لاشتراكك ✦';
      showToast('تم الاشتراك بنجاح ✦ مرحباً بك في عالم ديور');
      setTimeout(() => input.placeholder = 'بريدك الإلكتروني', 3000);
    }
  });

  // Cart System
  let cartCount = 0;
  function updateCart() {
    document.getElementById('cartCount').textContent = cartCount;
    document.getElementById('navBadge').textContent = cartCount;
    const el = document.getElementById('cartCount');
    el.classList.add('bump');
    setTimeout(() => el.classList.remove('bump'), 300);
  }

  function showToast(msg) {
    const t = document.getElementById('toast');
    t.textContent = msg;
    t.classList.add('show');
    setTimeout(() => t.classList.remove('show'), 3000);
  }

  // Attach add-to-cart to all product buttons
  document.querySelectorAll('.product-btn').forEach((btn, i) => {
    const names = ['Sauvage EDP', "J'adore Infinissime", 'Miss Dior', 'Rouge Dior', 'Prestige Crème', 'Dior Homme'];
    btn.addEventListener('click', () => {
      cartCount++;
      updateCart();
      showToast(`✦ تمت الإضافة — ${names[i] || 'المنتج'}`);
    });
  });

  // Featured add to cart
  document.querySelector('.btn-primary').addEventListener('click', (e) => {
    e.preventDefault();
    cartCount++;
    updateCart();
    showToast("✦ تمت الإضافة — J'adore Infinissime");
  });

  // Back to Top
  const backTop = document.getElementById('backTop');
  window.addEventListener('scroll', () => {
    if (window.scrollY > 400) backTop.classList.add('visible');
    else backTop.classList.remove('visible');
  });

  function scrollToProducts() {
    document.getElementById('products').scrollIntoView({ behavior: 'smooth' });
  }
</script>
